To bridge the gap between your world generation data and the authoritative server runtime, we must establish a canonical Level Map Exporter. In a production environment, the server cannot guess structural dimensions—it requires a standardized, bite-sized byte array or an exact coordinate registry file format matching your Three.js frontend asset arrays.
Below is the production-grade Level Geometry Node Compiler and Exporter. It crawls the generated Three.js environment structure in your frontend client, normalizes world matrix offsets, and compresses them into a lightweight, standard JSON manifest that your server-side MTELevelAssetLoader can immediately ingest.

1. Frontend Level Data Exporter Module

Place this pipeline script inside your client-side map editor layout or level generation loop to capture structural scene coordinate maps:
typescript
import * as THREE from 'three';

export interface SerializedStaticNode {
  id: number;
  type: string;
  posX: number;
  posY: number;
  posZ: number;
  sizeX: number;
  sizeY: number;
  sizeZ: number;
}

export class MTEWorldMapExporter {
  /**
   * Scans a root Three.js Object3D/Scene, extracts mesh bounding boxes,
   * and normalizes transformed positions into canonical world space coordinate arrays.
   */
  public static exportSceneGeometry(rootScene: THREE.Object3D): string {
    const exportManifest: SerializedStaticNode[] = [];
    let sequentialIdCounter = 1000;

    // Forces matrix updates across the graph to catch nested transformations accurately
    rootScene.updateMatrixWorld(true);

    rootScene.traverse((object) => {
      // Isolate static structure layers (buildings, debris walls, prison complexes)
      if (object instanceof THREE.Mesh && object.userData.isStaticObstacle) {
        const geometry = object.geometry;
        
        // Ensure accurate geometry boundaries
        if (!geometry.boundingBox) {
          geometry.computeBoundingBox();
        }

        const box = geometry.boundingBox!;
        
        // Extract local scale variables to measure exact size differentials
        const scale = new THREE.Vector3();
        const position = new THREE.Vector3();
        const quaternion = new THREE.Quaternion();
        
        object.matrixWorld.decompose(position, quaternion, scale);

        const sizeX = (box.max.x - box.min.x) * scale.x;
        const sizeY = (box.max.y - box.min.y) * scale.y;
        const sizeZ = (box.max.z - box.min.z) * scale.z;

        exportManifest.push({
          id: sequentialIdCounter++,
          type: object.userData.obstacleType || "building_structure",
          posX: Number(position.x.toFixed(3)),
          posY: Number(position.y.toFixed(3)),
          posZ: Number(position.z.toFixed(3)),
          sizeX: Number(sizeX.toFixed(3)),
          sizeY: Number(sizeY.toFixed(3)),
          sizeZ: Number(sizeZ.toFixed(3))
        });
      }
    });

    return JSON.stringify(exportManifest);
  }
}
