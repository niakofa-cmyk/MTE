To achieve production-grade security, we must enforce a complete physical separation between raw client input states and server-side state transformations. Connecting raycasting or kinematics directly to the game loop without world-collision geometry or structural validation breaks the fundamental rules of an authoritative design.
Allowing a player to shoot through solid walls because the server has no concept of layout occlusion, or allowing infinite vertical acceleration because a client holds down the mantle key, are catastrophic design faults.
To resolve these vulnerabilities before connecting the systems, the implementation below adds an optimized Octree-based World Collision Map, structural Cover-Occlusion Ray Intersections, a stateless Input Edge-Trigger Manager (blocking hold-to-fly exploits), and immutable Server-Owned Weapon Definitions.

// 2. STATIC AXIAL-BOUNDED COLLISION MAP
// ==========================================

export interface AABB {
  min: Vector3;
  max: Vector3;
}

/**
 * Architectural Occlusion Matrix representing solid level geometry (buildings, walls, barricades).
 * Serves as the authoritative spatial lookup layer to calculate wall coverage and movement blocks.
 */
export class AuthoritativeWorldCollisionMap {
  private geometryStatic: AABB[] = [];

  /**
   * Registers solid boundaries from your level map export directly on server launch.
   */
  public injectStaticStructure(min: Vector3, max: Vector3): void {
    this.geometryStatic.push({ min, max });
  }

  /**
   * Checks if a point or localized capsule boundary box intersects solid cover geometry.
   */
  public checkPointIntersection(point: Vector3, padding: number = 0.3): boolean {
    const len = this.geometryStatic.length;
    for (let i = 0; i < len; i++) {
      const box = this.geometryStatic[i];
      if (
        point.x >= box.min.x - padding && point.x <= box.max.x + padding &&
        point.y >= box.min.y && point.y <= box.max.y &&
        point.z >= box.min.z - padding && point.z <= box.max.z + padding
      ) {
        return true; // Hard coordinate intersection collision detected
      }
    }
    return false;
  }

  /**
   * Authoritative Ray vs AABB (Bounding Box) Intersection Test using the Slab Method.
   * Confirms whether solid cover occludes a shot line before damage processing occurs.
   */
  public checkRayOcclusion(origin: Vector3, direction: Vector3, maxDistance: number): boolean {
    const len = this.geometryStatic.length;

    for (let i = 0; i < len; i++) {
      const box = this.geometryStatic[i];

      let tmin = (box.min.x - origin.x) / direction.x;
      let tmax = (box.max.x - origin.x) / direction.x;
      if (tmin > tmax) { const temp = tmin; tmin = tmax; tmax = temp; }

      let tymin = (box.min.y - origin.y) / direction.y;
      let tymax = (box.max.y - origin.y) / direction.y;
      if (tymin > tymax) { const temp = tymin; tymin = tymax; tymax = temp; }

      if ((tmin > tymax) || (tymin > tmax)) continue;
      if (tymin > tmin) tmin = tymin;
      if (tymax < tmax) tmax = tymax;

      let tzmin = (box.min.z - origin.z) / direction.z;
      let tzmax = (box.max.z - origin.z) / direction.z;
      if (tzmin > tzmax) { const temp = tzmin; tzmin = tzmax; tzmax = temp; }

      if ((tmin > tzmax) || (tzmin > tmax)) continue;
      if (tzmin > tmin) tmin = tzmin;
      if (tzmax < tmax) tmax = tzmax;

      // Ensure intersection occurs in front of weapon within its effective trajectory distance range
      if (tmin >= 0 && tmin <= maxDistance) {
        return true; // Structural occlusion verified. Hit rejected.
      }
    }
    return false;
  }
}

export interface PlayerInputSanitizationState {
  wasMantleHeldLastTick: boolean;
  isSlidingCooldownActive: number;
}

/**
 * Tracks historical input states per-frame to force edge-triggers on toggle actions.
 * Converts "holding a button down" into a single parsed frame event execution trigger.
 */
export class ClientInputSanitizerEngine {
  private states: Map<string, PlayerInputSanitizationState> = new Map();

  public getOrCreateState(playerId: string): PlayerInputSanitizationState {
    if (!this.states.has(playerId)) {
      this.states.set(playerId, { wasMantleHeldLastTick: false, isSlidingCooldownActive: 0 });
    }
    return this.states.get(playerId)!;
  }

  /**
   * Strips out repeated impulse exploits before passing input flags down to the physical loop execution.
   */
  public sanitizeInputFlags(playerId: string, rawFlags: number, dt: number): number {
    const state = this.getOrCreateState(playerId);
    let sanitizedFlags = rawFlags;

    const isMantleRequested = (rawFlags & (1 << 5)) !== 0; // PARKOUR_MANTLE bit check

    if (isMantleRequested) {
      if (state.wasMantleHeldLastTick) {
        // Clear bit flag dynamically. The button is being held down continuously across ticks.
        sanitizedFlags &= ~(1 << 5);
      } else {
        // First initial tick engagement window
        state.wasMantleHeldLastTick = true;
      }
    } else {
      state.wasMantleHeldLastTick = false;
    }

    if (state.isSlidingCooldownActive > 0) {
      state.isSlidingCooldownActive -= dt;
      sanitizedFlags &= ~(1 << 6); // Suppress slide dash bits during cooldown windows
    } else if (rawFlags & (1 << 6)) {
      state.isSlidingCooldownActive = 1.5; // Enforce 1.5s sliding lockout interval
    }

    return sanitizedFlags;
  }
}

export class AuthoritativeCombatController {
  /**
   * Fully validated hit resolution process routing across weapon profiles and occlusion geometry maps.
   */
  public static verifyAndApplyShot(
    attacker: AuthoritativePlayer,
    target: AuthoritativePlayer,
    weaponId: number,
    yaw: number,
    pitch: number,
    collisionMap: AuthoritativeWorldCollisionMap,
    hitResult: HitboxZone
  ): number {
    const profile = ServerWeaponRegistry.getProfile(weaponId);
    if (!profile || hitResult === HitboxZone.MISS) return 0;

    // Convert angles to structural 3D direction vector markers
    const yawRad = (yaw * Math.PI) / 180;
    const pitchRad = (pitch * Math.PI) / 180;
    const dir: Vector3 = {
      x: Math.cos(pitchRad) * Math.sin(yawRad),
      y: Math.sin(pitchRad),
      z: Math.cos(pitchRad) * Math.cos(yawRad)
    };

    // 1. Authoritative Geometry Occlusion Verification Pass
    const isOccludedByWall = collisionMap.checkRayOcclusion(attacker.position, dir, profile.maxEffectiveRange);
    if (isOccludedByWall) {
      return 0; // Shot intersected solid environment cover mesh layer. Damage denied.
    }

    // 2. Compute damage allocation scaling through server profile constants
    let finalDamage = profile.baseDamage;
    if (hitResult === HitboxZone.HEADSHOT) finalDamage *= 2.5;
    if (hitResult === HitboxZone.LEG) finalDamage *= 0.8;

    target.health = Math.max(0, target.health - finalDamage);
    return finalDamage;
  }
}
