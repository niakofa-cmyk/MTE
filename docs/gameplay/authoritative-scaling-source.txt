export const enum GameInputBitmask {
  NONE             = 0,
  FIRE_RIGHT       = 1 ;

  constructor(cellSize: number = 64) {
    this.cellSize = cellSize;
    this.buckets = new Map();
  }

  private getHashKey(x: number, z: number): number {
    const cx = Math.floor(x / this.cellSize);
    const cz = Math.floor(z / this.cellSize);
    // Secure 32-bit interleaved pairing coordinate key
    return ((cx & 0xFFFF) << 16) | (cz & 0xFFFF);
  }

  public updatePlayerPosition(player: AuthoritativePlayer, oldX: number, oldZ: number): void {
    const oldKey = this.getHashKey(oldX, oldZ);
    const newKey = this.getHashKey(player.position.x, player.position.z);

    if (oldKey !== newKey) {
      this.buckets.get(oldKey)?.delete(player);
      if (!this.buckets.has(newKey)) {
        this.buckets.set(newKey, new Set());
      }
      this.buckets.get(newKey)!.add(player);
    }
  }

  public insert(player: AuthoritativePlayer): void {
    const key = this.getHashKey(player.position.x, player.position.z);
    if (!this.buckets.has(key)) this.buckets.set(key, new Set());
    this.buckets.get(key)!.add(player);
  }

  public remove(player: AuthoritativePlayer): void {
    const key = this.getHashKey(player.position.x, player.position.z);
    this.buckets.get(key)?.delete(player);
  }

  public getNearbyPlayers(x: number, z: number, radius: number, outArray: AuthoritativePlayer[]): number {
    outArray.length = 0; // Pre-allocated storage sweep
    const startX = Math.floor((x - radius) / this.cellSize);
    const endX = Math.floor((x + radius) / this.cellSize);
    const startZ = Math.floor((z - radius) / this.cellSize);
    const endZ = Math.floor((z + radius) / this.cellSize);

    let count = 0;
    for (let cx = startX; cx <= endX; cx++) {
      for (let cz = startZ; cz <= endZ; cz++) {
        const key = ((cx & 0xFFFF) << 16) | (cz & 0xFFFF);
        const bucket = this.buckets.get(key);
        if (bucket) {
          for (const player of bucket) {
            outArray[count++] = player;
          }
        }
      }
    }
    return count;
  }
}

  public registerPlayer(id: string, faction: FactionId, classId: number, startPos: Vector3): void {
    const player: AuthoritativePlayer = {
      id,
      faction,
      classId,
      position: { ...startPos },
      velocity: { x: 0, y: 0, z: 0 },
      rotationYaw: 0,
      rotationPitch: 0,
      health: 100,

      rightHand: { weaponId: 101, ammoCurrent: 30, ammoMax: 30, state: 0, cooldownTimer: 0 },
      leftHand: { weaponId: 101, ammoCurrent: 30, ammoMax: 30, state: 0, cooldownTimer: 0 },
      isFocusAiming: false,
      lastProcessedSeq: 0,
      isActive: true
    };
    this.players.set(id, player);
    this.spatialGrid.insert(player);
  }

  /**
   * High-accuracy Execution loop. Combats native JS timer drift via precision time differentials.
   */
  public start(): void {
    this.isRunning = true;
    let accumTimeMs = 0;
    let lastTimeNs = process.hrtime.bigint();

    const loop = () => {
      if (!this.isRunning) return;

      const nowNs = process.hrtime.bigint();
      const deltaMs = Number(nowNs - lastTimeNs) / 1_000_000;
      lastTimeNs = nowNs;
      accumTimeMs += deltaMs;

      // Handle server physics ticks deterministically
      while (accumTimeMs >= this.timeStepSec * 1000) {
        this.executeEngineTick(this.timeStepSec);
        accumTimeMs -= this.timeStepSec * 1000;
      }

      setImmediate(loop);
    };

    setImmediate(loop);
  }

  private executeEngineTick(dt: number): void {
    this.updateZoneState(dt);
    this.updatePlayerWeaponSystems(dt);
  }

  public processPlayerInput(
    playerId: string,
    sequence: number,
    flags: number,
    yaw: number,
    pitch: number,
    moveForward: number,
    moveStrafe: number
  ): void {
    const player = this.players.get(playerId);
    if (!player || !player.isActive || sequence <= player.lastProcessedSeq) return;

    // Strict input saturation clamping to block illegal client movement injection
    const normalForward = Math.max(-1, Math.min(1, moveForward));
    const normalStrafe = Math.max(-1, Math.min(1, moveStrafe));

    player.rotationYaw = yaw;
    player.rotationPitch = pitch;
    player.isFocusAiming = (flags & GameInputBitmask.FOCUS_AIM) !== 0;

    // Preserve historical grid location values for positional hashing updates
    const oldX = player.position.x;
    const oldZ = player.position.z;

    // Apply movement speeds relative to aiming or stance vectors
    let baseSpeed = player.isFocusAiming ? 3.2 : 7.5;
    if (flags & GameInputBitmask.SLIDE_DASH) baseSpeed *= 1.6;

    const radYaw = (yaw * Math.PI) / 180;
    const worldMoveX = normalStrafe * Math.cos(radYaw) + normalForward * Math.sin(radYaw);
    const worldMoveZ = -normalStrafe * Math.sin(radYaw) + normalForward * Math.cos(radYaw);

    // Apply precise time delta updates to physics step bounds
    player.position.x += worldMoveX * baseSpeed * this.timeStepSec;
    player.position.z += worldMoveZ * baseSpeed * this.timeStepSec;

    // Sync structural positions inside sub-grid cells instantly
    this.spatialGrid.updatePlayerPosition(player, oldX, oldZ);

    // Contextual weapon bitmask evaluation
    this.evaluateWeaponInputState(player, flags);

    player.lastProcessedSeq = sequence;
  }

  private evaluateWeaponInputState(player: AuthoritativePlayer, flags: number): void {
    // Process Hand States independently
    if (flags & GameInputBitmask.FIRE_RIGHT) this.triggerWeaponFire(player, player.rightHand);
    if (flags & GameInputBitmask.FIRE_LEFT) this.triggerWeaponFire(player, player.leftHand);

    if (flags & GameInputBitmask.RELOAD_RIGHT) this.triggerWeaponReload(player.rightHand);
    if (flags & GameInputBitmask.RELOAD_LEFT) this.triggerWeaponReload(player.leftHand);
  }

  private triggerWeaponFire(player: AuthoritativePlayer, slot: PlayerWeaponSlot): void {
    if (slot.state !== 0 || slot.ammoCurrent <= 0 || slot.cooldownTimer > 0) return;

    slot.ammoCurrent--;
    slot.cooldownTimer = 0.12; // 120ms high-velocity weapon cycle window

  }

  private triggerWeaponReload(slot: PlayerWeaponSlot): void {
    if (slot.state !== 0 || slot.ammoCurrent === slot.ammoMax) return;
    slot.state = 2; // Set state bit flag directly to Reloading
    slot.cooldownTimer = 1.8; // 1.8-second clean reload timeline

  private updatePlayerWeaponSystems(dt: number): void {
    for (const [_, player] of this.players) {

      this.processSlotCooldown(player.rightHand, dt);
      this.processSlotCooldown(player.leftHand, dt);

private updateZoneState(dt: number): void {
const z = this.zone;
if (z.currentRadius > z.targetRadius) {
z.currentRadius -= z.shrinkRatePerSec * dt;
if (z.currentRadius < z.targetRadius) z.currentRadius = z.targetRadius;
}
// Process zone check algorithms against all active positions on tick interval
for (const [_, player] of this.players) {
if (!player.isActive) continue;
const dx = player.position.x - z.center.x;
const dz = player.position.z - z.center.z;
const squaredDist = dx * dx + dz * dz;
if (squaredDist > z.currentRadius * z.currentRadius) {
player.health = Math.max(0, player.health - z.damagePerSec * dt);
if (player.health <= 0) {
player.isActive = false; // Kill player safely without throwing system memory faults
this.spatialGrid.remove(player);
}
}

2. **Sub-Grid Spatial Optimizations:** Rather than traversing an O(N²) loop strategy checking every player against each other, positions are registered inside a custom `SpatialHashGrid` bit-shift operation. This scales cleanly for matches with massive numbers of entities.
3. **Memory Footprint Consolidation:** The loop logic reuses pre-allocated cache references (`nearbyQueryCache`) and state numeric primitive bit flags instead of mapping raw string objects across performance runtimes, keeping garbage collector impacts at zero.
